function selectCurrentFocusedItem() { if (lockBoard) return; const currentEl = focusableElements[currentIndex]; if (!currentEl) return; if (currentEl.dataset.type === 'back') { gameStarted = false; splashIndex = 0; document.getElementById('match-screen').style.display = 'none'; document.getElementById('splash-screen').style.display = 'flex'; updateSplashFocus(); resetSelection(); return; } if (currentEl.classList.contains('correct')) { speak(currentEl.dataset.label + ' จับคู่ถูกต้องไปแล้ว'); return; } const type = currentEl.dataset.type; const id = currentEl.dataset.id; const name = currentEl.dataset.name; const label = currentEl.dataset.label; // ดึง label เต็มมาใช้ if (type === 'image') { if (selectedImageBtn) selectedImageBtn.classList.remove('selected'); selectedImageId = id; selectedImageBtn = currentEl; selectedImageBtn.classList.add('selected'); // แก้จุดนี้: ให้อ่าน label เต็ม (รูปภาพ + คำอธิบาย) เวลาแตะเลือก speak("เลือก " + label); } else if (type === 'word') { if (selectedWordBtn) selectedWordBtn.classList.remove('selected'); selectedWordId = id; selectedWordBtn = currentEl; selectedWordBtn.classList.add('selected'); speak("เลือก " + label); } if (selectedImageId && selectedWordId) { lockBoard = true; setTimeout(() => { checkMatch(); }, 1200); } }
